114/* Set one pixel after applying the configured rotation. */
115void Paint::DrawPixel(int x, int y, int colored) {
116 int point_temp;
117 if (this->rotate == ROTATE_0) {
118 if(x < 0 || x >= this->width || y < 0 || y >= this->height) {
119 return;
120 }
121 DrawAbsolutePixel(x, y, colored);
122 } else if (this->rotate == ROTATE_90) {
123 if(x < 0 || x >= this->height || y < 0 || y >= this->width) {
124 return;
125 }
126 point_temp = x;
127 x = this->width - y;
128 y = point_temp;
129 DrawAbsolutePixel(x, y, colored);
130 } else if (this->rotate == ROTATE_180) {
131 if(x < 0 || x >= this->width || y < 0 || y >= this->height) {
132 return;
133 }
134 x = this->width - x;
135 y = this->height - y;
136 DrawAbsolutePixel(x, y, colored);
137 } else if (this->rotate == ROTATE_270) {
138 if(x < 0 || x >= this->height || y < 0 || y >= this->width) {
139 return;
140 }
141 point_temp = x;
142 x = y;
143 y = this->height - point_temp;
144 DrawAbsolutePixel(x, y, colored);
145 }
146}

149void Paint::DrawCharAt(int x, int y, char ascii_char, sFONT* font, int colored) {
150 int i, j;
151 unsigned int char_offset = (ascii_char - ' ') * font->Height * (font->Width / 8 + (font->Width % 8 ? 1 : 0));
152 const unsigned char* ptr = &font->table[char_offset];
153
154 for (j = 0; j < font->Height; j++) {
155 for (i = 0; i < font->Width; i++) {
156 if (pgm_read_byte(ptr) & (0x80 >> (i % 8))) {
157 DrawPixel(x + i, y + j, colored);

159 if (i % 8 == 7) {
160 ptr++;
161 }

187/* Draw a line between two endpoints in the image buffer. */
188void Paint::DrawLine(int x0, int y0, int x1, int y1, int colored) {
189 /* Bresenham algorithm */
190 int dx = x1 - x0 >= 0 ? x1 - x0 : x0 - x1;
191 int sx = x0 < x1 ? 1 : -1;
192 int dy = y1 - y0 <= 0 ? y1 - y0 : y0 - y1;
193 int sy = y0 < y1 ? 1 : -1;
194 int err = dx + dy;
195
196 while((x0 != x1) && (y0 != y1)) {
197 DrawPixel(x0, y0 , colored);
198 if (2 * err >= dy) {
199 err += dy;
200 x0 += sx;
201 }
202 if (2 * err <= dx) {
203 err += dx;
204 y0 += sy;
205 }
206 }
207}

254void Paint::DrawCircle(int x, int y, int radius, int colored) {
255 /* Bresenham algorithm */
256 int x_pos = -radius;
257 int y_pos = 0;
258 int err = 2 - 2 * radius;
259 int e2;
260
261 do {
262 DrawPixel(x - x_pos, y + y_pos, colored);
263 DrawPixel(x + x_pos, y + y_pos, colored);
264 DrawPixel(x + x_pos, y - y_pos, colored);
265 DrawPixel(x - x_pos, y - y_pos, colored);
266 e2 = err;
267 if (e2 <= y_pos) {
268 err += ++y_pos * 2 + 1;
269 if(-x_pos == y_pos && e2 <= x_pos) {
270 e2 = 0;
271 }
272 }
273 if (e2 > x_pos) {
274 err += ++x_pos * 2 + 1;
275 }
276 } while (x_pos <= 0);
