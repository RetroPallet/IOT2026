135static int hex_nibble(char c) {
136 if (c >= '0' && c <= '9') return c - '0';
137 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
138 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
139 return -1;
140}

147static state_codes parse_downlink(const String &hex) {
148 if (hex.length() == 2) {
149 int hi = hex_nibble(hex[0]);
150 int lo = hex_nibble(hex[1]);
151 if (hi >= 0 && lo >= 0) {
152 int value = hi * 16 + lo;
153 if (value >= WAIT_INITIALIZATION_CODE && value <= COMPLETED_CODE)
154 return (state_codes)value;
155 }
156 }
157 Serial.print(F("Invalid downlink: "));
158 Serial.println(hex);
159 return NO_DOWNLINK_CODE;
160}
