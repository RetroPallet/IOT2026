150void game_task() {
151
152 epaper_task();
153
154 switch (state) {
155
156 case WAIT_INITIALIZATION:
157 if (button_pressed()) {
158 set_state(WAIT_BACKEND);
159 break;
160 }
161 break;
162
163 case PHASE_PUZZLE: {
164 if (button_pressed()) {
165 set_state(WAIT_BACKEND);
166 break;
167 }
168
169 bool phase_synced = phase_update();
170 bool temp_hot = temperature_triggered();
171 led_set(temp_hot ? LED_RED : LED_YELLOW);
172
173 if (phase_synced && temp_hot) {
174 if (!phase_solved) {
175 Serial.println(F("Phase puzzle solved!"));
176 phase_solved = true;
177 show_code(puzzle_codes[0]);
178 }
179 } else {
180 phase_solved = false;
181 }
182 break;
183 }
184
185 case SERIAL_PUZZLE:
186 if (button_pressed())
187 set_state(WAIT_BACKEND);
188 //show_employee_file(employee); commented for now check what implementation to pursue
189 if(password_update()){
190 // Serial.println(F("Password puzzle solved!\n"));
191 show_code(puzzle_codes[1]);
192 }
193 break;
194
195 case BACKEND_PUZZLE:
196 if (button_pressed())
197 set_state(WAIT_BACKEND);
198 break;
199
200 case COMPLETED:
201 // Raise flag
202 if (button_pressed())
203 set_state(WAIT_BACKEND);
204 break;
205
206 case WAIT_BACKEND: {
207 next_state_code = get_be_next_state_code();
208
209 switch (next_state_code) {
210 case WAIT_INITIALIZATION_CODE: next_state = WAIT_INITIALIZATION; break;
211 case PHASE_PUZZLE_CODE: next_state = PHASE_PUZZLE; break;
212 case SERIAL_PUZZLE_CODE: next_state = SERIAL_PUZZLE; break;
213 case BACKEND_PUZZLE_CODE: next_state = BACKEND_PUZZLE; break;
214 case COMPLETED_CODE: next_state = COMPLETED; break;
215 default: // NO_DOWNLINK / timeout
216 next_state = prvs_state;
217 break;
218 }
219
220 if (next_state != prvs_state) {
221 phase_solved = false;
222 reset_show_code();
223 }
224
225 set_state(next_state);
226 break;
227 }
228 }
