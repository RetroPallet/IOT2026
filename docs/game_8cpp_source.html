117void game_task() {
118
119 epaper_task();
120
121 switch (state) {
122
123 case WAIT_INITIALIZATION:
124 if (button_pressed()) {
125 set_state(WAIT_BACKEND);
126 break;
127 }
128 break;
129
130 case PHASE_PUZZLE: {
131 if (button_pressed()) {
132 set_state(WAIT_BACKEND);
133 break;
134 }
135
136 bool phase_synced = phase_update();
137 bool temp_hot = temperature_triggered();
138 led_set(temp_hot ? LED_RED : LED_YELLOW);
139
140 if (phase_synced && temp_hot) {
141 if (!phase_solved) {
142 Serial.println(F("Phase puzzle solved!"));
143 phase_solved = true;
144 show_code(puzzle_codes[0]);
145 }
146 } else {
147 phase_solved = false;
148 }
149 break;
150 }
151
152 case SERIAL_PUZZLE:
153 if (button_pressed())
154 set_state(WAIT_BACKEND);
155 if (password_update()) {
156 show_code(puzzle_codes[1]);
157 }
158 break;
159
160 case BACKEND_PUZZLE:
161 if (button_pressed())
162 set_state(WAIT_BACKEND);
163 break;
164
165 case COMPLETED:
166 if (button_pressed())
167 set_state(WAIT_BACKEND);
168 break;
169
170 case WAIT_BACKEND: {
171 next_state_code = get_be_next_state_code();
172
173 switch (next_state_code) {
174 case WAIT_INITIALIZATION_CODE: next_state = WAIT_INITIALIZATION; break;
175 case PHASE_PUZZLE_CODE: next_state = PHASE_PUZZLE; break;
176 case SERIAL_PUZZLE_CODE: next_state = SERIAL_PUZZLE; break;
177 case BACKEND_PUZZLE_CODE: next_state = BACKEND_PUZZLE; break;
178 case COMPLETED_CODE: next_state = COMPLETED; break;
179 default: // NO_DOWNLINK / timeout
180 next_state = prvs_state;
181 break;
182 }
183
184 if (next_state != prvs_state) {
185 phase_solved = false;
186 reset_show_code();
187 }
188
189 set_state(next_state);
190 break;
191 }
192 }
